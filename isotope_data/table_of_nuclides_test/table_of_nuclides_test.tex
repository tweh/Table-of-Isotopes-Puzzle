\begin{document}
\begin{tikzpicture}
  % axis
  \def\minZ{77}
  \def\maxZ{95}
  \def\minN{120}
  \def\maxN{148}
  \draw [->] (\minN-1,\minZ-1) -- (\maxN+1,\minZ-1) node [above] {\textbf{N}};
  \foreach \n in {\minN,...,\maxN}
    \draw (\n,\minZ-1) -- ++(0,-0.2) node [below] {\n};
  \draw [->] (\minN-1,\minZ-1) -- (\minN-1,\maxZ+1) node [right] {\textbf{Z}};
  \foreach \z in {\minZ,...,\maxZ}
    \draw (\minN-1,\z) -- ++(-0.2,0) node [left] {\z};
  % tiles
  \csvreader[
    separator = semicolon,
    head to column names,
    head to column names prefix = DATA,
  ]
  {../isotope_data.csv}{
    decay text=\DATAdecays
  }{
    \node at (\DATAN,\DATAZ) [tile node, \DATAcolor] {%
%      \textbf{\textsuperscript{\DATAA}\DATAsymbol}\\
      \textbf{\ce{^{\DATAA}_{\DATAZ}\DATAsymbol}}\\[0.5ex]
%      \DATAN,\DATAZ\\
      \footnotesize\DATAdecays
    };
  }
  % block lines
  \def\blockWidth{7}
  \def\blockHeight{7}
  \pgfmathsetmacro\nextN{\minN+\blockWidth}
  \pgfmathsetmacro\nextZ{\minZ+\blockHeight}
  \foreach \n in {\minN,\nextN,...,\maxN}
    \draw [block line] (\n-0.5,\minZ-0.5) -- (\n-0.5,\maxZ+1);
  \foreach \z in {\minZ,\nextZ,...,\maxZ}
    \draw [block line] (\minN-0.5,\z-0.5) -- (\maxN+1,\z-0.5);
\end{tikzpicture}
